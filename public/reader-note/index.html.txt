<script>
(function(){
  const KEY="reader-note.v1";
  const api=window.AiPhone;
  let state=load();
  let tab="home";

  function load(){
    try{
      const x=JSON.parse(localStorage.getItem(KEY)||"null");
      return x&&Array.isArray(x.items)?x:{items:[]};
    }catch(e){return {items:[]}}
  }
  function persist(){localStorage.setItem(KEY,JSON.stringify(state))}
  function esc(s){
    return String(s??"").replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[m]));
  }
  function toast(t){
    if(api&&api.ui&&api.ui.toast) api.ui.toast(t).catch(()=>{});
    else alert(t);
  }
  function openModal(item){
    document.getElementById("modal").classList.remove("hidden");
    document.getElementById("modalTitle").textContent=item?"编辑内容":"新建笔记";
    document.getElementById("fTitle").value=item?.title||"";
    document.getElementById("fUrl").value=item?.url||"";
    document.getElementById("fBody").value=item?.body||"";
    document.getElementById("fTags").value=(item?.tags||[]).join(" ");
    document.getElementById("saveBtn").dataset.id=item?.id||"";
  }
  function closeModal(){document.getElementById("modal").classList.add("hidden")}
  function save(){
    const title=document.getElementById("fTitle").value.trim()||"未命名笔记";
    const url=document.getElementById("fUrl").value.trim();
    const body=document.getElementById("fBody").value.trim();
    const tags=document.getElementById("fTags").value.trim().split(/\s+/).filter(Boolean);
    const id=document.getElementById("saveBtn").dataset.id;
    if(id){
      const x=state.items.find(i=>i.id===id);
      if(x) Object.assign(x,{title,url,body,tags,updated:Date.now()});
    }else{
      state.items.unshift({id:crypto.randomUUID(),type:url?"saved":"note",title,url,body,tags,created:Date.now(),updated:Date.now()});
    }
    persist();closeModal();render();
  }
  function remove(id){
    state.items=state.items.filter(x=>x.id!==id);persist();render();
  }
  function card(x){
    return `<article class="card">
      <div class="cardhead"><div><h3>${esc(x.title)}</h3><div class="meta">${new Date(x.updated||x.created).toLocaleString("zh-CN")}</div></div></div>
      ${x.url?`<div class="url">${esc(x.url)}</div>`:""}
      ${x.body?`<div style="margin-top:8px;white-space:pre-wrap">${esc(x.body.slice(0,280))}${x.body.length>280?"…":""}</div>`:""}
      ${(x.tags||[]).length?`<div class="tags">${x.tags.map(t=>`<span class="tag">#${esc(t)}</span>`).join("")}</div>`:""}
      <div class="actions">
        ${x.url?`<button class="btn" data-reader-url="${esc(x.url)}" onclick="window.__reader(this.dataset.readerUrl)">阅读</button>`:""}
        <button class="btn ghost" onclick="window.__edit('${x.id}')">编辑</button>
        <button class="btn danger" onclick="window.__remove('${x.id}')">删除</button>
      </div>
    </article>`;
  }
  function render(){
    const q=document.getElementById("search").value.trim().toLowerCase();
    let items=state.items.filter(x=>!q || [x.title,x.url,x.body,(x.tags||[]).join(" ")].join(" ").toLowerCase().includes(q));
    if(tab==="saved") items=items.filter(x=>x.type==="saved");
    if(tab==="notes") items=items.filter(x=>x.type==="note"||x.type==="clip");
    const c=document.getElementById("content");
    if(tab==="home"){
      c.innerHTML=`<div class="card"><h3>今天想记什么？</h3><p class="meta">把网页、摘录和灵感放进同一个地方。</p><div class="actions"><button class="btn" onclick="window.__openReader()">打开网页</button><button class="btn ghost" onclick="window.__newNote()">写一条笔记</button></div><small class="tip">收藏的网页只显示在「收藏」，新增的笔记只显示在「笔记」。</small></div>`;
    }else c.innerHTML=items.length?items.map(card).join(""):`<div class="empty">这里还没有内容</div>`;
  }

  window.__newNote=()=>openModal();
  window.__edit=id=>openModal(state.items.find(x=>x.id===id));
  window.__remove=remove;

  window.__openReader=()=>{
    document.getElementById("reader").classList.remove("hidden");
    document.getElementById("address").focus();
  };
  window.__reader=url=>{
    document.getElementById("reader").classList.remove("hidden");
    document.getElementById("address").value=url;
    document.getElementById("frame").src=normalize(url);
  };
  function normalize(u){return /^https?:\/\//i.test(u)?u:"https://"+u}
  document.getElementById("go").onclick=()=>{
    const u=document.getElementById("address").value.trim();
    if(!u)return;
    document.getElementById("frame").src=normalize(u);
  };
  document.getElementById("savePage").onclick=()=>{
    const u=document.getElementById("address").value.trim();
    if(!u)return;
    const title=u.replace(/^https?:\/\//,"").split("/")[0];
    const normalized=normalize(u);
    const existing=state.items.find(x=>x.type==="saved" && x.url===normalized);
    if(existing){
      toast("这个网页已经收藏");
    }else{
      state.items.unshift({id:crypto.randomUUID(),type:"saved",title,url:normalized,body:"",tags:["网页"],created:Date.now(),updated:Date.now()});
      persist();
      toast("网页已收藏");
    }
    render();
  };
  document.getElementById("quickNote").onclick=()=>{
    const u=document.getElementById("address").value.trim();
    openModal({url:normalize(u),tags:["摘录"]});
  };
  document.getElementById("closeReader").onclick=()=>document.getElementById("reader").classList.add("hidden");
  document.getElementById("newBtn").onclick=()=>openModal();
  document.getElementById("cancelBtn").onclick=closeModal;
  document.getElementById("saveBtn").onclick=save;
  document.getElementById("search").oninput=render;
  document.querySelectorAll(".tab").forEach(b=>b.onclick=()=>{
    tab=b.dataset.tab;
    document.querySelectorAll(".tab").forEach(x=>x.classList.toggle("on",x===b));
    render();
  });
  render();
})();
</script>
